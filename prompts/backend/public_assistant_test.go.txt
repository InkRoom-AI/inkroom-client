type publicTestCatalog struct {
	err   error
	calls int
}

func (p *publicTestCatalog) ListPublicPrompts(q promptapp.ListQuery) (promptapp.PromptList, error) {
	p.calls++
	if q.Scope != "" || q.FavoriteUserID != "" || q.PageSize > 15 {
		panic("non-public query")
	}
	return promptapp.PromptList{Items: []entity.Prompt{
		{ID: "coffee", Title: "咖啡馆海报", Prompt: "保留文字周末见", CoverURL: "/api/prompt-assets/coffee/012345678901234567890123.webp", GithubURL: "https://github.com/example/source"},
		{ID: "other", Title: "other", Prompt: "无关内容"},
	}}, p.err
}
func publicTestStore(t *testing.T) authguard.Store {
	t.Helper()
	db, err := gorm.Open(sqlite.Open(":memory:"), &gorm.Config{Logger: logger.Default.LogMode(logger.Silent)})
	if err != nil {
		t.Fatal(err)
	}
	sqlDB, _ := db.DB()
	sqlDB.SetMaxOpenConns(1)
	t.Cleanup(func() { _ = sqlDB.Close() })
	if err = db.AutoMigrate(&authguard.State{}); err != nil {
		t.Fatal(err)
	}
	return authguard.NewSQLStore(db)
}
func publicTestRequest(h *Handler, method, path, body string) *httptest.ResponseRecorder {
	gin.SetMode(gin.TestMode)
	router := gin.New()
	_ = router.SetTrustedProxies(nil)
	h.RegisterPublicRoutes(router.Group("/api"))
	request := httptest.NewRequest(method, path, strings.NewReader(body))
	request.RemoteAddr = "192.0.2.1:1234"
	request.Header.Set("Content-Type", "application/json")
	w := httptest.NewRecorder()
	router.ServeHTTP(w, request)
	return w
}
func TestPublicPromptSearchIsAnonymousAndReturnsOnlyPairedPublicData(t *testing.T) {
	catalog := &publicTestCatalog{}
	h := &Handler{publicAssistant: &publicAssistant{catalog: catalog, guard: publicTestStore(t)}}
	result := publicTestRequest(h, "GET", "/api/prompts/assistant/search?query=咖啡馆%20海报&scope=mine", "")
	if result.Code != 200 {
		t.Fatal(result.Code, result.Body.String())
	}
	var data struct {
		Items []publicPromptItem `json:"items"`
	}
	if json.Unmarshal(result.Body.Bytes(), &data) != nil || len(data.Items) != 1 || data.Items[0].Prompt != "保留文字周末见" || !strings.HasPrefix(data.Items[0].ExampleImageURL, "https://static.inkroom.ai/") {
		t.Fatal(result.Body.String())
	}
	if catalog.calls != 1 {
		t.Fatal(catalog.calls)
	}
}
func TestPublicPromptCatalogFailureDoesNotRunOptimizer(t *testing.T) {
	catalog := &publicTestCatalog{err: errors.New("private db detail")}
	called := false
	h := &Handler{publicAssistant: &publicAssistant{catalog: catalog, guard: publicTestStore(t), plan: func(string) (PublicPromptPlan, error) { called = true; return PublicPromptPlan{}, nil }}}
	result := publicTestRequest(h, "GET", "/api/prompts/assistant/search?query=咖啡馆", "")
	if result.Code != 503 || called || strings.Contains(result.Body.String(), "private db detail") {
		t.Fatal(result.Code, result.Body.String())
	}
}
func TestPublicPromptOptimizationStrictInputNoImageAndNoWallet(t *testing.T) {
	var calls int
	a := &publicAssistant{catalog: &publicTestCatalog{}, guard: publicTestStore(t), dailyBudget: 100, plan: func(query string) (PublicPromptPlan, error) {
		return PublicPromptPlan{Model: "text-model", ReservedMillis: 20, Run: func(context.Context) (string, error) { calls++; return "保留文字周末见", nil }}, nil
	}}
	h := &Handler{publicAssistant: a}
	for _, body := range []string{`{"query":"coffee","model":"image-model"}`, `{"query":"coffee"} {}`, `{"query":""}`, `{"query":"` + strings.Repeat("a", 5000) + `"}`} {
		if r := publicTestRequest(h, "POST", "/api/prompts/assistant/optimize", body); r.Code != 400 {
			t.Fatal(r.Code, r.Body.String())
		}
	}
	result := publicTestRequest(h, "POST", "/api/prompts/assistant/optimize", `{"query":"咖啡馆"}`)
	if result.Code != 200 || calls != 1 || !strings.Contains(result.Body.String(), `"exampleImageUrl":null`) {
		t.Fatal(result.Code, result.Body.String(), calls)
	}
}
func TestPublicPromptReservationSurvivesUnknownFailureAndNewHandler(t *testing.T) {
	store := publicTestStore(t)
	count := 0
	plan := func(string) (PublicPromptPlan, error) {
		return PublicPromptPlan{Model: "model", ReservedMillis: 60, Run: func(context.Context) (string, error) { count++; return "", errors.New("unknown provider result") }}, nil
	}
	a := &publicAssistant{catalog: &publicTestCatalog{}, guard: store, plan: plan, dailyBudget: 100}
	if r := publicTestRequest(&Handler{publicAssistant: a}, "POST", "/api/prompts/assistant/optimize", `{"query":"coffee"}`); r.Code != 502 {
		t.Fatal(r.Code)
	}
	b := &publicAssistant{catalog: &publicTestCatalog{}, guard: store, plan: plan, dailyBudget: 100}
	r := publicTestRequest(&Handler{publicAssistant: b}, "POST", "/api/prompts/assistant/optimize", `{"query":"coffee"}`)
	if r.Code != 429 || count != 1 || r.Header().Get("Retry-After") == "" {
		t.Fatal(r.Code, r.Body.String(), count)
	}
}
func TestPublicPromptSharedBudgetIsAtomicAcrossReplicas(t *testing.T) {
	store := publicTestStore(t)
	var accepted atomic.Int32
	var group sync.WaitGroup
	for i := 0; i < 20; i++ {
		group.Add(1)
		go func(i int) {
			defer group.Done()
			a := &publicAssistant{guard: store, dailyBudget: 100}
			if a.reserve(context.Background(), "192.0.2."+strconv.Itoa(i+1), true, 30) == nil {
				accepted.Add(1)
			}
		}(i)
	}
	group.Wait()
	if accepted.Load() != 3 {
		t.Fatal("budget admits", accepted.Load())
	}
}
