package prompt

import (
	domainrepo "cloud-canvas/internal/domain/prompt/repository"
	"github.com/glebarez/sqlite"
	"gorm.io/gorm"
	"testing"
)

func TestPublicPromptKeywordsAreLiteralANDAndApplyBeforePagination(t *testing.T) {
	db, err := gorm.Open(sqlite.Open(":memory:"), &gorm.Config{})
	if err != nil {
		t.Fatal(err)
	}
	if err = db.AutoMigrate(&AssetObject{}, &Prompt{}); err != nil {
		t.Fatal(err)
	}
	rows := []Prompt{
		{ID: "both", Title: "咖啡馆", Prompt: "开业 海报 100% x_y", Tags: []string{"暖色"}, Enabled: true},
		{ID: "subject-only", Title: "咖啡馆人像", Prompt: "无文字", Enabled: true},
		{ID: "wrong-percent", Title: "咖啡馆海报", Prompt: "1000 xyz", Enabled: true},
	}
	if err = db.Create(&rows).Error; err != nil {
		t.Fatal(err)
	}
	repo := NewRepository(db)
	for _, terms := range [][]string{{"咖啡馆", "海报", "100%"}, {"咖啡馆", "x_y"}, {"海报", "暖色"}} {
		items, total, err := repo.ListPrompts(domainrepo.PromptQuery{Keywords: terms, Page: 1, PageSize: 1})
		if err != nil || total != 1 || len(items) != 1 || items[0].ID != "both" {
			t.Fatalf("terms=%v total=%d items=%v err=%v", terms, total, items, err)
		}
	}
	items, total, err := repo.ListPrompts(domainrepo.PromptQuery{Keyword: "咖啡馆", Page: 1, PageSize: 5})
	if err != nil || total != 3 || len(items) != 3 {
		t.Fatal("legacy keyword behavior changed", total, err)
	}
}
