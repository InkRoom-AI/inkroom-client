package ai

import (
	"context"
	"io"
	"net/http"
	"net/http/httptest"
	"sync/atomic"
	"testing"
	"time"

	apikeyentity "cloud-canvas/internal/domain/apikey/entity"
)

// This real TCP upstream reads the full request before hanging. It performs no
// model work and charges nothing; cancellation must reach its request context.
func publicPromptHangingUpstream(t *testing.T) (string, <-chan struct{}, <-chan struct{}, *atomic.Int32) {
	t.Helper()
	started, cancelled := make(chan struct{}, 1), make(chan struct{}, 1)
	release := make(chan struct{})
	calls := &atomic.Int32{}
	server := httptest.NewServer(http.HandlerFunc(func(_ http.ResponseWriter, r *http.Request) {
		calls.Add(1)
		if _, err := io.Copy(io.Discard, r.Body); err != nil {
			t.Errorf("controlled upstream could not read the request: %v", err)
			return
		}
		select {
		case started <- struct{}{}:
		default:
		}
		select {
		case <-r.Context().Done():
			select {
			case cancelled <- struct{}{}:
			default:
			}
		case <-release:
		}
	}))
	t.Cleanup(server.Close)
	// Release before Close even if an assertion fails, avoiding a stuck test.
	t.Cleanup(func() { close(release) })
	return server.URL, started, cancelled, calls
}

func TestPublicPromptInFlightHTTPCancellationReachesUpstreamWithoutRetry(t *testing.T) {
	url, started, cancelled, calls := publicPromptHangingUpstream(t)
	h := publicPromptTestHandler(t, url)
	repo := attachImageLogUsageRecorder(t, h)
	plan, err := h.PreparePublicPrompt("a controlled cancellation fixture")
	if err != nil {
		t.Fatal(err)
	}
	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()
	done := make(chan error, 1)
	go func() {
		_, err := plan.Run(ctx)
		done <- err
	}()
	select {
	case <-started:
		cancel()
	case <-time.After(3 * time.Second):
		t.Fatal("request never reached the controlled upstream")
	}
	select {
	case err := <-done:
		if err == nil {
			t.Fatal("cancelled request was delivered as success")
		}
	case <-time.After(3 * time.Second):
		t.Fatal("in-flight request did not return after cancellation")
	}
	select {
	case <-cancelled:
	case <-time.After(3 * time.Second):
		t.Fatal("cancellation did not reach the upstream TCP request")
	}
	item := publicPromptInternalUsage(t, repo)
	if calls.Load() != 1 || item.Status != apikeyentity.UsageStatusCancelled || item.HTTPStatus != 0 || item.InputTokens != 0 || item.OutputTokens != 0 || item.CachedTokens != 0 || item.RequestParameters != `{"usage_reported":false}` {
		t.Fatalf("cancelled attempt retried, lost its record or estimated usage: calls=%d item=%#v", calls.Load(), item)
	}
}

func TestPublicPromptFixedThirtySecondHTTPTimeoutReachesUpstreamWithoutRetry(t *testing.T) {
	url, started, cancelled, calls := publicPromptHangingUpstream(t)
	h := publicPromptTestHandler(t, url)
	repo := attachImageLogUsageRecorder(t, h)
	plan, err := h.PreparePublicPrompt("a controlled fixed-timeout fixture")
	if err != nil {
		t.Fatal(err)
	}
	type result struct {
		text    string
		err     error
		elapsed time.Duration
	}
	done := make(chan result, 1)
	go func() {
		start := time.Now()
		// No caller deadline: this exercises the production http.Client's fixed
		// 30-second timeout, not a shorter test-only context or transport fake.
		text, err := plan.Run(context.Background())
		done <- result{text: text, err: err, elapsed: time.Since(start)}
	}()
	select {
	case <-started:
	case <-time.After(3 * time.Second):
		t.Fatal("request never reached the controlled upstream")
	}
	select {
	case got := <-done:
		if got.err == nil || got.text != "" || got.elapsed < 30*time.Second || got.elapsed > 34*time.Second {
			t.Fatalf("fixed 30-second timeout was not enforced: text=%q err=%v elapsed=%s", got.text, got.err, got.elapsed)
		}
	case <-time.After(35 * time.Second):
		t.Fatal("production client did not enforce its fixed timeout")
	}
	select {
	case <-cancelled:
	case <-time.After(3 * time.Second):
		t.Fatal("client timeout did not close the upstream TCP request")
	}
	item := publicPromptInternalUsage(t, repo)
	if calls.Load() != 1 || item.Status != apikeyentity.UsageStatusFailed || item.HTTPStatus != 0 || item.DurationMillis < 30_000 || item.DurationMillis > 34_000 || item.InputTokens != 0 || item.OutputTokens != 0 || item.CachedTokens != 0 || item.RequestParameters != `{"usage_reported":false}` {
		t.Fatalf("timed-out attempt retried, lost its record or estimated usage: calls=%d item=%#v", calls.Load(), item)
	}
}
